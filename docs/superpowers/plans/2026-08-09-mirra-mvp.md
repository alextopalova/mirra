# Mirra MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Mirra — an in-store smart-mirror kiosk web app that diagnoses a shopper's personal color + body type from photos, recommends matched store garments, and lets them virtually try on the top pick via YouCam Apparel VTO.

**Architecture:** Two services. A **React + TypeScript + Vite** frontend (liquid-glass kiosk UI, camera capture, VTO viewer) talks to a **FastAPI (Python)** backend that owns the body-CV classifier (MediaPipe Pose + NumPy), a recommendation scorer pipeline over a curated catalog, and a proxy for the YouCam APIs (key stays server-side).

**Tech Stack:** Frontend — React 18, TypeScript, Vite, plain CSS (design tokens), MediaDevices camera API. Backend — Python 3.11, FastAPI, Uvicorn, MediaPipe, OpenCV, NumPy, httpx, Pydantic v2, pytest. Deploy — frontend on Vercel/Netlify, backend on Render/Railway (or local for kiosk demo).

## Global Constraints

- **Build priority (overriding):** good-looking + fully-functioning end-to-end beats feature depth. Front-load UI + mock click-through, then wire real pieces. Simplify/stub anything that threatens "looks great + works."
- **CV is pragmatic, not SOTA:** classifier works from **front photo + height + weight** (MediaPipe Pose ratios). Side photo is **optional** enhancement. **No SAM/SHAPY.**
- **YouCam try-on is real and central** — never stub it in the final demo.
- **YouCam API key is server-side only.** Never shipped to the client. Loaded from env `YOUCAM_API_KEY` (+ secret if required).
- **YouCam free budget = 1,000 units.** Cache results per session; use mocked responses in dev; only call live during real runs.
- **Kiosk UI:** grey/blue liquid glass, white-on-dark; touch targets ≥72px; on-screen numeric keypad for height/weight; pose-guide overlay on capture; results screen is an intermediate stop with one dominant "Try it on" CTA.
- **Frontend has zero CV** — it only uploads photos + inputs and renders results.
- **Commit after every task.**

```tsx
import { createContext, useContext, useState } from "react";
import type { Screen, BodyProfile, Palette, Recommendation } from "../api/types";

interface SessionData {
  frontPhoto?: string;  // dataURL
  sidePhoto?: string;
  heightCm?: number;
  weightKg?: number;
  profile?: BodyProfile;
  palette?: Palette;
  category?: string;
  occasion?: string;
  recommendations?: Recommendation[];
  selectedId?: string;
  tryOnImage?: string;  // dataURL/URL from VTO
}

interface Ctx {
  screen: Screen;
  data: SessionData;
  go: (s: Screen) => void;
  update: (patch: Partial<SessionData>) => void;
  reset: () => void;
}

const SessionCtx = createContext<Ctx | null>(null);

export function SessionProvider({ children }: { children: React.ReactNode }) {
  const [screen, setScreen] = useState<Screen>("start");
  const [data, setData] = useState<SessionData>({});
  const go = (s: Screen) => setScreen(s);
  const update = (patch: Partial<SessionData>) => setData((d) => ({ ...d, ...patch }));
  const reset = () => { setData({}); setScreen("start"); };
  return <SessionCtx.Provider value={{ screen, data, go, update, reset }}>{children}</SessionCtx.Provider>;
}

export function useSession() {
  const c = useContext(SessionCtx);
  if (!c) throw new Error("useSession outside provider");
  return c;
}
```

```tsx
import { useEffect, useRef, useState } from "react";
import { useSession } from "../state/session";
import { PrimaryButton } from "../components/PrimaryButton";
import { PoseGuide } from "../components/PoseGuide";
import "./capture.css";
import "./screen.css";

export function CaptureScreen() {
  const { go, update } = useSession();
  const videoRef = useRef<HTMLVideoElement>(null);
  const [step, setStep] = useState<"front" | "side">("front");
  const [count, setCount] = useState<number | null>(null);

  useEffect(() => {
    let stream: MediaStream;
    navigator.mediaDevices.getUserMedia({ video: { facingMode: "user" } }).then((s) => {
      stream = s;
      if (videoRef.current) videoRef.current.srcObject = s;
    });
    return () => stream?.getTracks().forEach((t) => t.stop());
  }, []);

  const snap = (): string => {
    const v = videoRef.current!;
    const c = document.createElement("canvas");
    c.width = v.videoWidth; c.height = v.videoHeight;
    c.getContext("2d")!.drawImage(v, 0, 0);
    return c.toDataURL("image/jpeg", 0.9);
  };

  const runCountdown = (after: () => void) => {
    let n = 3; setCount(n);
    const id = setInterval(() => {
      n -= 1;
      if (n === 0) { clearInterval(id); setCount(null); after(); } else setCount(n);
    }, 700);
  };

  const capture = () => runCountdown(() => {
    const img = snap();
    if (step === "front") { update({ frontPhoto: img }); setStep("side"); }
    else { update({ sidePhoto: img }); go("measurements"); }
  });

  return (
    <div className="screen">
      <h2>{step === "front" ? "Face the camera" : "Turn to your side"}</h2>
      <div className="capture-stage">
        <video ref={videoRef} autoPlay playsInline muted />
        <PoseGuide variant={step} />
        {count !== null && <div className="capture-countdown">{count}</div>}
      </div>
      <p className="capture-hint">Stand so your whole body fits the outline.</p>
      <div style={{ display: "flex", gap: 16 }}>
        <PrimaryButton label={step === "front" ? "Capture front" : "Capture side"} onClick={capture} />
        {step === "side" && (
          <PrimaryButton label="Skip side" variant="ghost" onClick={() => go("measurements")} />
        )}
      </div>
    </div>
  );
}
```

export function NumericKeypad({ value, onChange, max = 3 }: {
  value: string; onChange: (next: string) => void; max?: number;
}) {
  const press = (d: string) => { if (value.length < max) onChange(value + d); };
  const back = () => onChange(value.slice(0, -1));
  return (
    <div className="keypad">
      {["1","2","3","4","5","6","7","8","9"].map((d) => (
        <button key={d} onClick={() => press(d)}>{d}</button>
      ))}
      <button onClick={back}>⌫</button>
      <button onClick={() => press("0")}>0</button>
      <button onClick={() => onChange("")}>C</button>
    </div>
  );
}

```python
from app.schemas import BodyMeasurements, BodyProfile

def _fruit(m: BodyMeasurements) -> str:
    shoulder, waist, hip, bust = m.shoulder_w, m.waist_w, m.hip_w, m.bust_w
    whr = waist / hip if hip else 1.0
    sh_hip = shoulder / hip if hip else 1.0
    defined_waist = whr <= 0.80

    # Apple: midsection is the widest point (often with higher BMI).
    if waist >= hip and waist >= bust and m.bmi >= 26:
        return "apple"
    if sh_hip >= 1.07:
        return "inverted-triangle"
    if sh_hip <= 0.93:
        return "pear"
    # shoulders ≈ hips → hourglass vs rectangle depends on waist definition
    return "hourglass" if defined_waist else "rectangle"

def _japanese_weights(m: BodyMeasurements) -> dict[str, float]:
    # Heuristic scores for Straight / Wave / Natural, then normalized.
    upper_heavy = m.torso_len < m.leg_len          # center of gravity
    depth = m.torso_depth if (m.has_side and m.torso_depth) else None
    whr = m.waist_w / m.hip_w if m.hip_w else 1.0

    straight = 0.0
    wave = 0.0
    natural = 0.0

    # Straight: upper-heavy, thicker torso depth, firmer (higher-ish BMI), less waist gap
    straight += 1.0 if not upper_heavy is False else 0.0
    straight += 1.0 if m.bmi >= 23 else 0.3
    if depth is not None:
        straight += 1.2 if depth >= 0.45 else 0.0   # thick front-to-back
    straight += 0.8 if whr >= 0.78 else 0.2

    # Wave: lower-heavy, thin depth, defined waist, softer (lower BMI)
    wave += 1.0 if upper_heavy is False else 0.0     # lower-heavy => torso_len >= leg? handled below
    wave += 1.0 if m.torso_len >= m.leg_len * 0.72 and m.leg_len >= m.torso_len else 0.0
    wave += 1.0 if whr <= 0.78 else 0.2
    wave += 0.8 if m.bmi < 23 else 0.2
    if depth is not None:
        wave += 1.2 if depth < 0.45 else 0.0         # flat front-to-back

    # Natural: broad frame / shoulders, balanced, bonier (lower BMI, longer limbs)
    natural += 1.0 if (m.shoulder_w / m.hip_w if m.hip_w else 1) >= 1.02 else 0.2
    natural += 0.8 if m.leg_len >= m.torso_len * 1.4 else 0.2
    natural += 0.8 if m.bmi < 22 else 0.2

    scores = {"straight": max(straight, 0.01), "wave": max(wave, 0.01), "natural": max(natural, 0.01)}
    total = sum(scores.values())
    return {k: round(v / total, 4) for k, v in scores.items()}

_LEAN = {"straight": "Straight", "wave": "Wave", "natural": "Natural"}

def classify(m: BodyMeasurements) -> BodyProfile:
    fruit = _fruit(m)
    weights = _japanese_weights(m)
    japanese = max(weights, key=weights.get)
    confidence = round(weights[japanese], 3)
    summary = f"{fruit.replace('-', ' ').title()} · {_LEAN[japanese]}-leaning"
    return BodyProfile(
        fruit=fruit, japanese=japanese, japanese_weights=weights,
        confidence=confidence, summary=summary,
    )
```

```python
import cv2
import numpy as np
import mediapipe as mp
from app.schemas import BodyMeasurements

_mp_pose = mp.solutions.pose

def _landmarks(bgr):
    with _mp_pose.Pose(static_image_mode=True, model_complexity=2) as pose:
        res = pose.process(cv2.cvtColor(bgr, cv2.COLOR_BGR2RGB))
    if not res.pose_landmarks:
        raise ValueError("no person detected")
    h, w = bgr.shape[:2]
    return [(lm.x * w, lm.y * h) for lm in res.pose_landmarks.landmark]

L = _mp_pose.PoseLandmark

def measure_from_images(front_bgr, side_bgr=None, height_cm=None, weight_kg=None) -> BodyMeasurements:
    p = _landmarks(front_bgr)
    def dx(a, b): return abs(p[a][0] - p[b][0])
    def dy(a, b): return abs(p[a][1] - p[b][1])

    shoulder_w = dx(L.LEFT_SHOULDER, L.RIGHT_SHOULDER)
    hip_w = dx(L.LEFT_HIP, L.RIGHT_HIP)
    # bust ≈ slightly below shoulders; waist ≈ midpoint shoulder→hip. Approximate from spans.
    bust_w = shoulder_w * 0.92
    waist_w = (shoulder_w + hip_w) / 2 * 0.82   # cinch factor; refined by segmentation later
    shoulder_y = (p[L.LEFT_SHOULDER][1] + p[L.RIGHT_SHOULDER][1]) / 2
    hip_y = (p[L.LEFT_HIP][1] + p[L.RIGHT_HIP][1]) / 2
    ankle_y = (p[L.LEFT_ANKLE][1] + p[L.RIGHT_ANKLE][1]) / 2
    torso_len = abs(hip_y - shoulder_y)
    leg_len = abs(ankle_y - hip_y)

    bmi = 22.0
    if height_cm and weight_kg:
        m2 = (height_cm / 100.0) ** 2
        bmi = round(weight_kg / m2, 1)

    torso_depth = None
    has_side = False
    if side_bgr is not None:
        try:
            sp = _landmarks(side_bgr)
            # front-to-back depth ≈ horizontal span at torso in the side view, normalized by torso height
            depth_px = abs(sp[L.LEFT_HIP][0] - sp[L.LEFT_SHOULDER][0]) + \
                       abs(sp[L.LEFT_SHOULDER][0] - sp[L.NOSE][0]) * 0.0
            side_torso = abs(sp[L.LEFT_HIP][1] - sp[L.LEFT_SHOULDER][1]) or 1.0
            torso_depth = round(min(depth_px / side_torso, 1.0), 3)
            has_side = True
        except ValueError:
            pass

    return BodyMeasurements(
        shoulder_w=shoulder_w, bust_w=bust_w, waist_w=waist_w, hip_w=hip_w,
        torso_len=torso_len, leg_len=leg_len, bmi=bmi,
        has_side=has_side, torso_depth=torso_depth,
    )
```

> Implementer note: waist/bust are approximated from landmark spans (Pose has no waist landmark). If time allows, refine by measuring the silhouette mask width at the waist y-level (MediaPipe Selfie Segmentation). Good enough for MVP per the build-priority constraint.

```python
import math
from app.reco.catalog import Garment
from app.schemas import BodyProfile

def _dE(a, b):
    return math.sqrt(sum((x - y) ** 2 for x, y in zip(a, b)))

def color_score(garment_lab, palette_labs) -> float:
    if not palette_labs:
        return 0.5
    best = min(_dE(garment_lab, p) for p in palette_labs)   # CIE76 distance
    return round(max(0.0, 1.0 - best / 60.0), 3)            # 60 ≈ "very different"

# Japanese-type preferences: fabric / structure / waist emphasis.
_JP_RULES = {
    "straight": {"structured": True,  "fabric": {"crisp", "structured", "smooth"}, "note": "clean structured lines suit Straight"},
    "wave":     {"structured": False, "fabric": {"soft", "draping", "light"},      "note": "soft high-waist shapes suit Wave"},
    "natural":  {"structured": False, "fabric": {"textured", "natural", "chunky"}, "note": "relaxed textured pieces suit Natural"},
}
# Fruit balance: which garment traits flatter each shape.
_FRUIT_NOTE = {
    "pear": "adds interest up top to balance hips",
    "apple": "elongates and skims the midsection",
    "hourglass": "defines the waist",
    "rectangle": "creates curve and waist definition",
    "inverted-triangle": "adds volume below to balance shoulders",
}

def body_score(g: Garment, profile: BodyProfile) -> tuple[float, list[str]]:
    reasons: list[str] = []
    score = 0.4
    rule = _JP_RULES[profile.japanese]
    if g.silhouette.get("structured") == rule["structured"]:
        score += 0.25
    if g.silhouette.get("fabric") in rule["fabric"]:
        score += 0.2; reasons.append(rule["note"])
    if profile.fruit in {"hourglass", "rectangle", "wave"} and g.silhouette.get("waist") == "defined":
        score += 0.15; reasons.append(f"Defines the waist ({profile.japanese.title()})")
    if not reasons:
        reasons.append(_FRUIT_NOTE.get(profile.fruit, "flatters your shape"))
    return round(min(score, 1.0), 3), reasons

def occasion_score(g: Garment, occasion: str) -> float:
    return 1.0 if occasion in g.occasion_tags else 0.6
```

from app.reco.scorers import color_score, body_score, occasion_score

def rank(profile, palette_labs, category, occasion, catalog):
    out = []
    for g in catalog:
        if g.category != category:
            continue
        cs = color_score(g.color_lab, palette_labs)
        bs, reasons = body_score(g, profile)
        os_ = occasion_score(g, occasion)
        score = round(0.45 * cs + 0.4 * bs + 0.15 * os_, 3)
        top_reasons = ([f"{profile.__dict__.get('palette_season','Your')} color match"] if cs > 0.7 else []) + reasons
        out.append({"garment": g, "score": score, "reasons": top_reasons[:2] or ["A strong match"]})
    out.sort(key=lambda r: r["score"], reverse=True)
    return out

from pydantic import BaseModel
from app.reco.catalog import load_catalog
from app.reco.engine import rank
from app.schemas import BodyProfile

router = APIRouter()

class Palette(BaseModel):
    season: str
    colors: list[str]

class RecommendIn(BaseModel):
    profile: BodyProfile
    palette: Palette
    category: str
    occasion: str

def _hex_to_lab(hex_str: str) -> list[float]:
    import colorsys
    h = hex_str.lstrip("#")
    r, g, b = (int(h[i:i+2], 16) / 255 for i in (0, 2, 4))
    # sRGB → XYZ → Lab (D65)
    def inv(c): return ((c + 0.055) / 1.055) ** 2.4 if c > 0.04045 else c / 12.92
    r, g, b = inv(r), inv(g), inv(b)
    x = (r*0.4124 + g*0.3576 + b*0.1805) / 0.95047
    y = (r*0.2126 + g*0.7152 + b*0.0722)
    z = (r*0.0193 + g*0.1192 + b*0.9505) / 1.08883
    def f(t): return t ** (1/3) if t > 0.008856 else 7.787 * t + 16/116
    fx, fy, fz = f(x), f(y), f(z)
    return [round(116*fy - 16, 2), round(500*(fx-fy), 2), round(200*(fy-fz), 2)]

@router.post("/recommend")
def recommend(inp: RecommendIn):
    palette_labs = [_hex_to_lab(c) for c in inp.palette.colors]
    catalog = load_catalog()
    recs = rank(inp.profile, palette_labs, inp.category, inp.occasion, catalog)
    return [{"garment": r["garment"].model_dump(), "score": r["score"], "reasons": r["reasons"]} for r in recs]

```python
import asyncio
import httpx
from app.config import settings

BASE = "https://yce-api-01.perfectcorp.com"   # replace with the base from CONTRACT.md

class YouCamClient:
    def __init__(self, transport: httpx.BaseTransport | None = None):
        self._client = httpx.AsyncClient(base_url=BASE, transport=transport, timeout=60)

    def _headers(self):
        # Replace per CONTRACT.md (Bearer vs key+secret token).
        return {"Authorization": f"Bearer {settings.youcam_api_key}"}

    async def upload(self, image_bytes: bytes) -> str:
        # Replace path/field per CONTRACT.md.
        r = await self._client.post("/s2s/v1.0/file", headers=self._headers(),
                                    files={"file": ("img.jpg", image_bytes, "image/jpeg")})
        r.raise_for_status()
        return r.json()["file_id"]

    async def run(self, task_type: str, payload: dict) -> str:
        r = await self._client.post(f"/s2s/v1.0/task/{task_type}", headers=self._headers(), json=payload)
        r.raise_for_status()
        return r.json()["task_id"]

    async def poll(self, task_id: str, interval: float = 1.5, max_tries: int = 40) -> dict:
        for _ in range(max_tries):
            r = await self._client.get(f"/s2s/v1.0/task/{task_id}", headers=self._headers())
            r.raise_for_status()
            data = r.json()
            if data.get("status") in ("done", "success", "completed"):
                return data
            if data.get("status") in ("error", "failed"):
                raise RuntimeError(f"YouCam task failed: {data}")
            await asyncio.sleep(interval)
        raise TimeoutError("YouCam task timed out")
```
